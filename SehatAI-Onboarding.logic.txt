\nclass Component extends DCLogic {\n  state = { idx: 0, email: '', done: false, answers: { confidence: null, challenges: [], goal: null, time: null } };\n\n  flow = ['welcome', 'confidence', 'challenges', 'goal', 'time'];\n\n  cfg = {\n    confidence: {\n      q: 'How confident are you with AI at work?',\n      hint: 'Pick the one that stings least — no wrong answer.',\n      short: 'Confidence',\n      multi: false,\n      options: [\n        { value: 'beginner', label: 'Beginner', desc: \"New to this. I've barely touched AI in my workflow.\" },\n        { value: 'some', label: 'Some experience', desc: \"I've tried it. Results are hit or miss.\" },\n        { value: 'confident', label: 'Confident', desc: 'I use it daily — I just want the edge cases.' },\n      ],\n    },\n    challenges: {\n      q: 'Where does your day leak the most time?',\n      hint: 'Choose all that apply. We aim the lessons here first.',\n      short: 'Challenges',\n      multi: true,\n      options: [\n        { value: 'docs', label: 'Documentation & notes', desc: 'Visit notes, charting, the endless typing.' },\n        { value: 'summaries', label: 'Chart & record summaries', desc: 'Pulling the signal out of long histories.' },\n        { value: 'comms', label: 'Patient communication', desc: 'Portal replies, education, difficult news.' },\n        { value: 'coord', label: 'Team coordination', desc: 'Handoffs, follow-ups, herding humans.' },\n        { value: 'prompteng', label: 'Prompt engineering', desc: 'Getting AI to actually do what you mean.' },\n      ],\n    },\n    goal: {\n      q: 'What do you want out of SehatAI?',\n      hint: 'One anchor goal — you can chase the rest later.',\n      short: 'Goal',\n      multi: false,\n      options: [\n        { value: 'confident', label: 'Use AI confidently day-to-day', desc: 'Stop guessing, start trusting your process.' },\n        { value: 'time', label: 'Save time on documentation', desc: 'Win back the hours the keyboard steals.' },\n        { value: 'prompts', label: 'Write better prompts', desc: 'Ask sharper, get answers worth using.' },\n        { value: 'verify', label: 'Verify AI output safely', desc: 'Catch the confident nonsense before it lands.' },\n        { value: 'compliant', label: 'Stay compliant & private', desc: 'Know what is safe to share, and what never is.' },\n        { value: 'cme', label: 'Earn CME credits', desc: 'Turn daily practice into logged credit.' },\n      ],\n    },\n    time: {\n      q: 'How much time can you give it a day?',\n      hint: 'Daily beats long. Be honest — streaks are the point.',\n      short: 'Cadence',\n      multi: false,\n      options: [\n        { value: '2', label: '2 minutes', desc: 'A scenario card. The micro-dose.' },\n        { value: '5', label: '5 minutes', desc: 'A lesson bite. The sweet spot.' },\n        { value: '10', label: '10 minutes', desc: 'A full interactive case. Go deep.' },\n      ],\n    },\n  };\n\n  leftCopy = {\n    welcome: { h: 'Your shield against confident nonsense.', b: 'SehatAI coaches clinicians to use AI safely, confidently, and on the job — built around real tasks, not abstract theory.' },\n    confidence: { h: 'We meet you where you are.', b: 'The path calibrates to your actual level — not where a generic course assumes you should be.' },\n    challenges: { h: 'Point us at the pain.', b: 'Tell us where the day leaks time, and the first lessons will aim straight at it.' },\n    goal: { h: 'Name the win.', b: 'One goal anchors your path and shapes what we surface each day.' },\n    time: { h: 'Small, but daily.', b: 'A few minutes a day compounds. Pick a dose you will actually keep.' },\n    done: { h: \"You're all set, Doctor.\", b: 'Your path is calibrated. Time to earn your first streak.' },\n  };\n\n  stepLabels = ['Sign in', 'Confidence', 'Challenges', 'Goal', 'Cadence'];\n\n  select(key, value, multi) {\n    this.setState(s => {\n      const a = { ...s.answers };\n      if (multi) {\n        a[key] = a[key].includes(value) ? a[key].filter(x => x !== value) : [...a[key], value];\n      } else {\n        a[key] = value;\n      }\n      return { answers: a };\n    });\n  }\n\n  canContinue() {\n    if (this.state.done) return true;\n    const key = this.flow[this.state.idx];\n    if (key === 'welcome') return this.state.email.trim().length > 3;\n    const c = this.cfg[key];\n    return c.multi ? this.state.answers[key].length > 0 : !!this.state.answers[key];\n  }\n\n  next() {\n    if (!this.canContinue()) return;\n    this.setState(s => {\n      if (s.idx >= this.flow.length - 1) return { done: true };\n      return { idx: s.idx + 1 };\n    });\n  }\n\n  back() {\n    this.setState(s => {\n      if (s.done) return { done: false };\n      return { idx: Math.max(0, s.idx - 1) };\n    });\n  }\n\n  cardStyle(selected) {\n    return {\n      display: 'flex', alignItems: 'flex-start', gap: '14px', textAlign: 'left', width: '100%',\n      cursor: 'pointer', padding: '18px 20px', borderRadius: 'var(--radius-lg)', font: 'inherit',\n      background: selected ? 'var(--color-accent-100)' : 'var(--color-neutral-100)',\n      border: selected ? '2px solid var(--color-accent)' : '2px solid transparent',\n      boxShadow: 'var(--shadow-sm)', transition: 'border-color .15s, background .15s',\n    };\n  }\n\n  dotStyle(selected) {\n    return {\n      flex: 'none', width: '24px', height: '24px', borderRadius: '50%', display: 'grid', placeContent: 'center',\n      fontSize: '14px', fontWeight: '700', marginTop: '2px',\n      background: selected ? 'var(--color-accent)' : 'transparent',\n      color: selected ? '#fff' : 'transparent',\n      border: selected ? '2px solid var(--color-accent)' : '2px solid var(--color-neutral-300)',\n    };\n  }\n\n  renderVals() {\n    const { idx, done, email, answers } = this.state;\n    const key = done ? 'done' : this.flow[idx];\n    const isWelcome = key === 'welcome';\n    const isDone = done;\n    const isQuestion = !isWelcome && !isDone;\n    const c = isQuestion ? this.cfg[key] : null;\n\n    const optionCards = c ? c.options.map(o => {\n      const sel = c.multi ? answers[key].includes(o.value) : answers[key] === o.value;\n      return {\n        label: o.label, desc: o.desc, selected: sel,\n        style: this.cardStyle(sel), dot: this.dotStyle(sel), check: sel ? '✓' : '',\n        onClick: () => this.select(key, o.value, c.multi),\n      };\n    }) : [];\n\n    const stepPos = done ? this.flow.length : idx;\n    const stepList = this.stepLabels.map((label, i) => {\n      const active = !done && i === idx;\n      const complete = done || i < idx;\n      return {\n        label,\n        badge: complete ? '✓' : String(i + 1),\n        dotStyle: {\n          flex: 'none', width: '28px', height: '28px', borderRadius: '50%', display: 'grid', placeContent: 'center',\n          fontSize: '13px', fontWeight: '700',\n          background: complete ? 'var(--color-accent)' : active ? 'var(--color-bg)' : 'transparent',\n          color: complete ? '#fff' : active ? 'var(--color-accent-800)' : 'color-mix(in srgb, var(--color-text) 45%, transparent)',\n          border: active ? '2px solid var(--color-accent)' : complete ? '2px solid var(--color-accent)' : '2px solid color-mix(in srgb, var(--color-text) 22%, transparent)',\n        },\n        labelStyle: {\n          fontSize: '14.5px', fontWeight: active ? '700' : '500',\n          color: active || complete ? 'var(--color-text)' : 'color-mix(in srgb, var(--color-text) 50%, transparent)',\n        },\n      };\n    });\n\n    const recap = [];\n    if (isDone) {\n      const cl = this.cfg.confidence.options.find(o => o.value === answers.confidence);\n      const gl = this.cfg.goal.options.find(o => o.value === answers.goal);\n      const tl = this.cfg.time.options.find(o => o.value === answers.time);\n      const chs = this.cfg.challenges.options.filter(o => answers.challenges.includes(o.value)).map(o => o.label);\n      recap.push({ k: 'Confidence', v: cl ? cl.label : '—' });\n      recap.push({ k: 'Focus areas', v: chs.length ? chs.join(', ') : '—' });\n      recap.push({ k: 'Anchor goal', v: gl ? gl.label : '—' });\n      recap.push({ k: 'Daily cadence', v: tl ? tl.label + ' / day' : '—' });\n    }\n\n    const cont = this.canContinue();\n    const continueStyle = {\n      borderRadius: '999px', paddingLeft: '28px', paddingRight: '24px',\n      opacity: cont ? '1' : '.45', pointerEvents: cont ? 'auto' : 'none',\n    };\n\n    const lc = this.leftCopy[key] || this.leftCopy.welcome;\n\n    return {\n      leftHeading: lc.h, leftBody: lc.b, stepList,\n      progressStyle: { height: '100%', borderRadius: '999px', background: 'var(--color-accent)', width: `${Math.round((stepPos / this.flow.length) * 100)}%`, transition: 'width .3s' },\n      stepLabel: isDone ? 'Complete' : `Step ${idx + 1} of ${this.flow.length}`,\n      isWelcome, isQuestion, isDone,\n      question: c ? c.q : '', hint: c ? c.hint : '',\n      optionCards,\n      email, onEmail: e => this.setState({ email: e.target.value }),\n      onSSO: () => this.setState(s => ({ email: s.email || 'clinician@hospital.org', idx: Math.min(this.flow.length - 1, s.idx + 1) })),\n      onNext: () => this.next(), onBack: () => this.back(),\n      continueStyle, continueLabel: idx >= this.flow.length - 1 ? 'Finish' : 'Continue',\n      backStyle: { borderRadius: '999px', visibility: (isWelcome || isDone) ? 'hidden' : 'visible' },\n      showFooter: !isWelcome && !isDone,\n      recap,\n    };\n  }\n}\n
